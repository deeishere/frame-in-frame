""" Frame in Frame — hand-tracked live effect (OpenCV + MediaPipe) ================================================================ Make a rectangle with both hands (thumb + index finger of each hand) and whatever inner clip/image you point this at gets composited live inside that hand-frame, on top of your webcam feed. Spread your hands wide enough and it crossfades to fill the whole screen — a built-in "reveal" transition. INSTALL ------- pip install opencv-python mediapipe numpy RUN --- python hand_frame_in_frame.py # placeholder pattern inside the frame python hand_frame_in_frame.py --inner clip.mp4 # video inside the frame python hand_frame_in_frame.py --inner photo.jpg # image inside the frame python hand_frame_in_frame.py --camera 1 # use a different webcam python hand_frame_in_frame.py --no-mirror # don't mirror the feed CONTROLS (while the window is focused) --------------------------------------- q quit s save a snapshot of the current composited frame (PNG) m toggle manual mode (drag 4 corner dots with the mouse instead of using your hands — also the automatic fallback if MediaPipe can't find hands well in your lighting/camera) r reset manual corner positions """ import argparse import time import os from datetime import datetime import cv2 import numpy as np import mediapipe as mp # -------------------------------------------------------------------------- # Geometry helpers # -------------------------------------------------------------------------- def sort_as_polygon(points): """Order 4 points into a simple (non-self-intersecting) polygon by sweeping angle around their centroid.""" pts = np.array(points, dtype=np.float32) cx, cy = pts[:, 0].mean(), pts[:, 1].mean() angles = np.arctan2(pts[:, 1] - cy, pts[:, 0] - cx) order = np.argsort(angles) return pts[order] def lerp_polygon(prev, target, t): """Smooth prev -> target, matching point i to point i (both already sorted the same way) to avoid corner-swap jitter.""" return prev + (target - prev) * t def polygon_area(pts): x = pts[:, 0] y = pts[:, 1] return 0.5 * abs(np.dot(x, np.roll(y, 1)) - np.dot(y, np.roll(x, 1))) def cover_crop_resize(src, target_w, target_h): """Resize+crop src to exactly fill (target_w, target_h), like CSS object-fit: cover.""" src_h, src_w = src.shape[:2] scale = max(target_w / src_w, target_h / src_h) new_w, new_h = int(np.ceil(src_w * scale)), int(np.ceil(src_h * scale)) resized = cv2.resize(src, (new_w, new_h), interpolation=cv2.INTER_LINEAR) x0 = (new_w - target_w) // 2 y0 = (new_h - target_h) // 2 return resized[y0:y0 + target_h, x0:x0 + target_w] def placeholder_tile(w, h, t): """Generated diagonal-stripe placeholder used when no --inner file is given, so the effect is visible out of the box.""" tile = np.zeros((h, w, 3), dtype=np.uint8) tile[:] = (18, 16, 16) # BGR near-black stripe_w = 18 shift = int(t * 40) % (stripe_w * 2) for x in range(-h - shift, w, stripe_w * 2): pts = np.array([ [x, h], [x + h, 0], [x + h + stripe_w, 0], [x + stripe_w, h] ], dtype=np.int32) cv2.fillConvexPoly(tile, pts, (196, 224, 82)) # BGR teal-ish overlay = tile.copy() cv2.addWeighted(overlay, 0.18, np.zeros_like(tile), 0.82, 0, tile) return tile # -------------------------------------------------------------------------- # Inner content source (image or looping video) # -------------------------------------------------------------------------- class InnerSource: def __init__(self, path): self.kind = None self.image = None self.cap = None if path is None: self.kind = "placeholder" return ext = os.path.splitext(path)[1].lower() if ext in (".jpg", ".jpeg", ".png", ".bmp", ".webp"): img = cv2.imread(path) if img is None: raise FileNotFoundError(f"Could not read image: {path}") self.image = img self.kind = "image" else: cap = cv2.VideoCapture(path) if not cap.isOpened(): raise FileNotFoundError(f"Could not open video: {path}") self.cap = cap self.kind = "video" def get_frame(self, w, h, t): if self.kind == "image": return cover_crop_resize(self.image, w, h) if self.kind == "video": ok, frame = self.cap.read() if not ok: # loop self.cap.set(cv2.CAP_PROP_POS_FRAMES, 0) ok, frame = self.cap.read() if not ok: return placeholder_tile(w, h, t) return cover_crop_resize(frame, w, h) return placeholder_tile(w, h, t) # -------------------------------------------------------------------------- # Manual mode (mouse-dragged corners) — also the automatic fallback # -------------------------------------------------------------------------- class ManualCorners: def __init__(self, w, h): self.w, self.h = w, h self.points = self._default() self.dragging = -1 def _default(self): w, h = self.w, self.h return np.array([ [w * 0.28, h * 0.30], [w * 0.72, h * 0.30], [w * 0.72, h * 0.72], [w * 0.28, h * 0.72], ], dtype=np.float32) def reset(self): self.points = self._default() def mouse_cb(self, event, x, y, flags, userdata): if event == cv2.EVENT_LBUTTONDOWN: dists = np.linalg.norm(self.points - np.array([x, y]), axis=1) idx = int(np.argmin(dists)) if dists[idx] < 35: self.dragging = idx elif event == cv2.EVENT_MOUSEMOVE and self.dragging >= 0: self.points[self.dragging] = [x, y] elif event == cv2.EVENT_LBUTTONUP: self.dragging = -1 # -------------------------------------------------------------------------- # Main # -------------------------------------------------------------------------- def main(): parser = argparse.ArgumentParser(description="Hand-tracked frame-in-frame effect") parser.add_argument("--inner", type=str, default=None, help="Path to an image or video to show inside the hand frame") parser.add_argument("--camera", type=int, default=0, help="Webcam index") parser.add_argument("--width", type=int, default=1280) parser.add_argument("--height", type=int, default=720) parser.add_argument("--reveal-start", type=float, default=0.12, help="Frame-area ratio (0-1) at which the crossfade starts kicking in") parser.add_argument("--reveal-threshold", type=float, default=0.55, help="Frame-area ratio (0-1) at which the inner content fully takes over the screen") parser.add_argument("--no-mirror", action="store_true", help="Don't mirror the webcam feed") args = parser.parse_args() mirror = not args.no_mirror mp_hands = mp.solutions.hands hands = mp_hands.Hands( max_num_hands=2, model_complexity=1, min_detection_confidence=0.6, min_tracking_confidence=0.5, ) cap = cv2.VideoCapture(args.camera) cap.set(cv2.CAP_PROP_FRAME_WIDTH, args.width) cap.set(cv2.CAP_PROP_FRAME_HEIGHT, args.height) if not cap.isOpened(): raise RuntimeError(f"Could not open camera index {args.camera}") ok, frame = cap.read() if not ok: raise RuntimeError("Could not read from camera") h, w = frame.shape[:2] inner = InnerSource(args.inner) manual = ManualCorners(w, h) window = "Frame in Frame" cv2.namedWindow(window) cv2.setMouseCallback(window, manual.mouse_cb) manual_mode = False smoothed = None reveal_progress = 0.0 t0 = time.time() print("Controls: q=quit s=snapshot m=toggle manual mode r=reset manual corners") while True: ok, frame = cap.read() if not ok: break if mirror: frame = cv2.flip(frame, 1) h, w = frame.shape[:2] t = time.time() - t0 target_points = None if manual_mode: target_points = manual.points else: rgb = cv2.cvtColor(frame, cv2.COLOR_BGR2RGB) results = hands.process(rgb) hand_pts = {} if results.multi_hand_landmarks and results.multi_handedness: for hand_landmarks, handedness in zip(results.multi_hand_landmarks, results.multi_handedness): # MediaPipe's handedness label assumes a mirrored # (selfie-view) input image. If we're not mirroring, # the label is backwards relative to the user, so flip it. label = handedness.classification[0].label if not mirror: label = "Right" if label == "Left" else "Left" thumb = hand_landmarks.landmark[4] index = hand_landmarks.landmark[8] hand_pts[label] = { "thumb": [thumb.x * w, thumb.y * h], "index": [index.x * w, index.y * h], } if "Left" in hand_pts and "Right" in hand_pts: # Fixed corner roles (regardless of hand rotation): # top-left = left index, top-right = right index # bottom-right = right thumb, bottom-left = left thumb target_points = np.array([ hand_pts["Left"]["index"], hand_pts["Right"]["index"], hand_pts["Right"]["thumb"], hand_pts["Left"]["thumb"], ], dtype=np.float32) if target_points is not None: target_sorted = np.array(target_points, dtype=np.float32) if smoothed is None: smoothed = target_sorted else: # Corner roles are now fixed (by hand/fingertip identity), so # points keep the same order every frame - no re-sorting needed. t_lerp = 1.0 if manual_mode else 0.35 smoothed = lerp_polygon(smoothed, target_sorted, t_lerp) elif not manual_mode: smoothed = None # release frame when hands drop out output = frame.copy() if smoothed is not None: poly = smoothed poly_int = poly.astype(np.int32) x0, y0, rect_w, rect_h = cv2.boundingRect(poly_int) rect_w, rect_h = max(rect_w, 1), max(rect_h, 1) x0 = max(x0, 0) y0 = max(y0, 0) x1 = min(x0 + rect_w, w) y1 = min(y0 + rect_h, h) inner_patch = inner.get_frame(x1 - x0, y1 - y0, t) mask = np.zeros((h, w), dtype=np.uint8) cv2.fillConvexPoly(mask, poly_int, 255) canvas = frame.copy() canvas[y0:y1, x0:x1] = inner_patch mask_3 = cv2.cvtColor(mask, cv2.COLOR_GRAY2BGR) output = np.where(mask_3 == 255, canvas, output) # viewfinder overlay: dashed polygon + corner dots cv2.polylines(output, [poly_int], isClosed=True, color=(196, 224, 82), thickness=2) for p in poly_int: cv2.circle(output, tuple(p), 6, (196, 224, 82), -1) area = polygon_area(poly) ratio = area / (w * h) # Continuous crossfade: fully driven by how wide you spread the # frame, so how much of the screen the inner content takes over # is entirely up to you, not a hard on/off snap. span = max(args.reveal_threshold - args.reveal_start, 1e-6) target_reveal = np.clip((ratio - args.reveal_start) / span, 0.0, 1.0) reveal_progress += (target_reveal - reveal_progress) * 0.15 if reveal_progress > 0.01: full_inner = inner.get_frame(w, h, t) output = cv2.addWeighted(full_inner, reveal_progress, output, 1 - reveal_progress, 0) hud = f"{'MANUAL' if manual_mode else 'TRACK'} FRAME {ratio*100:4.0f}%" if reveal_progress > 0.02: hud += f" REVEAL {reveal_progress*100:3.0f}%" cv2.putText(output, hud, (12, h - 16), cv2.FONT_HERSHEY_SIMPLEX, 0.55, (196, 224, 82), 1, cv2.LINE_AA) else: reveal_progress *= 0.9 msg = "Show both hands (thumb + index touching) to open a frame" if manual_mode: msg = "Drag the four dots to shape your frame" cv2.putText(output, msg, (12, h - 16), cv2.FONT_HERSHEY_SIMPLEX, 0.55, (150, 150, 150), 1, cv2.LINE_AA) cv2.putText(output, "Frame in Frame", (12, 28), cv2.FONT_HERSHEY_SIMPLEX, 0.6, (232, 230, 225), 1, cv2.LINE_AA) cv2.imshow(window, output) key = cv2.waitKey(1) & 0xFF if key == ord('q'): break elif key == ord('m'): manual_mode = not manual_mode print("Manual mode:", manual_mode) elif key == ord('r'): manual.reset() elif key == ord('s'): fname = f"frame_in_frame_{datetime.now().strftime('%Y%m%d_%H%M%S')}.png" cv2.imwrite(fname, output) print("Saved", fname) cap.release() if inner.cap is not None: inner.cap.release() cv2.destroyAllWindows() if __name__ == "__main__": main()
